	@RequestMapping("/loginsubmit.htm")
	public ModelAndView loginSubmit(HttpServletRequest request, HttpServletResponse response)
			throws ServletException, IOException {

		System.out.println("=== LOGIN CONTROLLER CALLED 2 ===");
		logger.info("loginSubmit(HttpServletRequest request, HttpServletResponse response) method begins");
		HttpSession session = request.getSession();
		session.setAttribute("refereshFlag_1", new Boolean(true));

		if (servletFlag) {
			servletFlag = false;
		}

		String userIPaddress = request.getHeader("ClientIP");
		if (request.getRequestURL().toString().contains("localhost")) {
			userIPaddress = "127.0.0.1";
		}
		SBIModelViewCache cache = null;

		Map outParams = new HashMap();
		String userName = request.getParameter(Constants.USER_NAME);
		String password = request.getParameter(Constants.PASSWORD);

		UserAgent userAgent = UserAgent.parseUserAgentString(request.getHeader("User-Agent"));
		String userInitial = "";
		
		outParams.put("keyString", request.getParameter("keyString"));
		String keyString = (String) request.getAttribute("keyString");
		
		if (userName != null && userName.trim() != "" && password != null && password.trim() != "") {

			// ==========CAPTCHA VALIDATION FIRST=========
			String captchaInput = request.getParameter("captchaInput");
			String sessionCaptcha = (session != null) ? (String) session.getAttribute("captcha") : null;

			if (sessionCaptcha == null || captchaInput == null || !captchaInput.equals(sessionCaptcha)) {
				logger.warn("CAPTCHA validation failed for user: " + userName);

				ApplicationResponse failureResponse = new ApplicationResponse(Constants.FAILURE);
				failureResponse.setErrorCode("CAPTCHA_ERROR");

				outParams.put(Constants.APPLICATION_RESPONSE, failureResponse);
				outParams.put(Constants.ERROR_VIEW, "loginDisplay");
				outParams.put(Constants.USER_NAME, userName);

				// Generate new keyString for the return page
				SecureRandom rand = new SecureRandom();
				String keyString2 = rand.toString();
				outParams.put("keyString", keyString2);

				// Clear the captcha from session
				session.removeAttribute("captcha");

				return new ModelAndView("loginDisplay", Constants.ERROR_MODEL, outParams);
			}

			// Clear captcha after successful validation
			session.removeAttribute("captcha");
			logger.info("CAPTCHA validation successful");
			// ========== CAPTCHA VALIDATION - END ==========

			session.setAttribute("bulkEmailEnabled", CommonDAOImpl.isBulkEmailEnabled());

			// ========== STEP 2: USERNAME & PASSWORD VALIDATION (ONLY AFTER CAPTCHA IS VALID) ==========
			Map inputParams = new HashMap();
			inputParams.put(Constants.USER_NAME, userName); 
			inputParams.put(Constants.PASSWORD, password);
			inputParams.put(Constants.IP_ADDRESS, userIPaddress);
			inputParams.put("sessionId", session.getId());
			inputParams.put("firstLogin", request.getParameter("firstLogin"));
			inputParams.put("keyId", request.getParameter("keyString"));
			inputParams.put("type", "login");

			outParams = (Map) logonService.execute(inputParams);
			outParams.put(Constants.USER_NAME, userName);
			outParams.put(Constants.PASSWORD, password);
			outParams.put(Constants.IP_ADDRESS, userIPaddress);

			// Check if user exists
			if ("NO".equals(outParams.get("UserExists"))) {
				logger.info("User Doesn't Exist :: Authentication failed and redirecting to login page");
				ApplicationResponse failureResponse = new ApplicationResponse(Constants.FAILURE);
//				failureResponse.setErrorCode("UserNotActive");
				failureResponse.setErrorCode("UserNotExist");
				outParams.put(Constants.APPLICATION_RESPONSE, failureResponse);
				outParams.put(Constants.ERROR_VIEW, "loginDisplay");
				
				// Generate new keyString for the return page
				SecureRandom rand = new SecureRandom();
				String keyString2 = rand.toString();
				outParams.put("keyString", keyString2);
				
				return new ModelAndView("loginDisplay", Constants.ERROR_MODEL, outParams);
			}

			// Check if password is wrong
			ApplicationResponse appRsp = (ApplicationResponse) outParams.get(Constants.APPLICATION_RESPONSE);
			if (appRsp != null && appRsp.getStatus().equalsIgnoreCase(Constants.FAILURE)) {
				logger.info("Invalid username/password :: Authentication failed");
				outParams.put(Constants.ERROR_VIEW, "loginDisplay");
				
				// Generate new keyString for the return page
				SecureRandom rand = new SecureRandom();
				String keyString2 = rand.toString();
				outParams.put("keyString", keyString2);
				
				return new ModelAndView("loginDisplay", Constants.ERROR_MODEL, outParams);
			}

			// ========== REST OF CODE CONTINUES HERE (Session creation, etc.) ==========
			Map<String, Object> temp = new ConcurrentHashMap<String, Object>();
			Enumeration e = session.getAttributeNames();
			while (e != null && e.hasMoreElements()) {
				String name = (String) e.nextElement();
				Object value = session.getAttribute(name);
				temp.put(name, value);
			}
			session.invalidate();
			HttpSession session1 = request.getSession();

			for (Map.Entry<String, Object> stringObjectEntry : temp.entrySet()) {
				session1.setAttribute(stringObjectEntry.getKey(), stringObjectEntry.getValue());
			}

			session1.setAttribute("outParams", outParams);

			if (null != request.getParameter("tokenId") && !("".equalsIgnoreCase(request.getParameter("tokenId")))) {
				if (null != cache.getOldToken() || !("".equalsIgnoreCase(cache.getOldToken()))) {
					cache.setOldToken(request.getParameter("tokenId").toString());
				}
			}

			cache = session1.getAttribute("cachevar") == null ? new SBIModelViewCache()
					: (SBIModelViewCache) session1.getAttribute("cachevar");

			session1.setAttribute("cachevar", cache);
			session1.setAttribute("userName", userName);

			cache.setOldToken((String) request.getParameter("tokenId"));

			if ("YES".equals(outParams.get("UserExists")) && (Integer) outParams.get("testFlag") > 0) {

				session1.setAttribute("ForgotFlag", 0);
				ApplicationResponse appRsp2 = (ApplicationResponse) outParams.get(Constants.APPLICATION_RESPONSE);
				session1.setAttribute(Constants.USER, outParams.get(Constants.USER));
				request.getRequestDispatcher("/changeloginpassword.htm").forward(request, response);
			} else {

				ApplicationResponse appRsp2 = (ApplicationResponse) outParams.get(Constants.APPLICATION_RESPONSE);
				if (appRsp2.getStatus().equalsIgnoreCase(Constants.SUCCESS)) {
					session1.setAttribute(Constants.USER, outParams.get(Constants.USER));
					User user = (User) outParams.get(Constants.USER);
					userInitial = (String) user.getUserInitial();

					if (user.getUserId() != null) {
						logger.info("User Authenticated and redirecting to landing page");
						session1.setAttribute("FAG", "ANT1");
						request.setAttribute("auditAction", "LOGIN");
						request.setAttribute("dataString", "");
						response.sendRedirect(baseURL + myPageURL.get(user.getType()));
						System.out.println("=== DEBUG INFO ===");
						System.out.println("baseURL: " + baseURL);
						System.out.println("user.getType(): " + user.getType());
						System.out.println("myPageURL.get(user.getType()): " + myPageURL.get(user.getType()));
						System.out.println("==================");

						return null;
					}
					outParams.put(Constants.ERROR_VIEW, "loginDisplay");
				} else {
					outParams.put("keyString", request.getParameter("keyString"));
					logger.info("loginSubmit(HttpServletRequest request, HttpServletResponse response) method ends");
					return new ModelAndView("loginDisplay", Constants.ERROR_MODEL, outParams);
				}
			}
		}
		logger.info("loginSubmit(HttpServletRequest request, HttpServletResponse response) method ends");

		return new ModelAndView("loginDisplay", Constants.ERROR_MODEL, outParams);
	}

	@RequestMapping("/logout.htm")
	public ModelAndView logoutHandler(HttpServletRequest request, HttpServletResponse response) {
		logger.info("logoutHandler(HttpServletRequest request, HttpServletResponse response) method begins");
		HttpSession session = request.getSession(false);
		Map outParams = new HashMap();
		if (session != null) {
			User user = (User) session.getAttribute(Constants.USER);
			if (user != null) {
				CommonDAOImpl.removeLoggdinUser(user.getUserAlias());

				if (user.getRole().equals("ADMIN"))
					outParams.put("role", "admin");
			}

			session.invalidate();

		}
		SecureRandom rand = new SecureRandom();
		String keyString = rand.toString();
		outParams.put("keyString", keyString);
		outParams.put("logout", "logout");
		request.setAttribute("auditAction", "LOGOUT");
		request.setAttribute("dataString", "");
		logger.info("logoutHandler(HttpServletRequest request, HttpServletResponse response) method ends");
		return new ModelAndView("logoutdisplay", Constants.ERROR_MODEL, outParams);
	}

	public ModelAndView forgetpasswordsubmit(HttpServletRequest request, HttpServletResponse response) {
		logger.info("forgetpasswordsubmit(HttpServletRequest request, HttpServletResponse response) method begins");
		HttpSession session = request.getSession(true);
		Map inParams = new HashMap();
		Map outParams = new HashMap();
		Enumeration params = request.getParameterNames();
		while (params.hasMoreElements()) {
			String param = (String) params.nextElement();
			inParams.put(param, request.getParameter(param));
		}

		String captcha = "";
		captcha = request.getParameter("captcha");
		// Added by Koteswara for CAPTCHA Starts

		if (captcha == null || (!captcha.equalsIgnoreCase("success"))) {

			/* Map<Object,Object> outParams = new HashMap<Object,Object>(); */
			String keyId = ""; // added for JSEncryption
			// return_view = getReturnView(bankCode);
			String contextName = request.getContextPath().substring(1);
			captcha = outParams.get("captcha").toString();
			if ("failure".equalsIgnoreCase(captcha)) {
				String userName = request.getParameter("");
				outParams.put("userName", userName);
				return new ModelAndView("FpassDisplay", "errorModel", outParams);
			}
		}

		inParams.put("type", "forgetpassword");
		outParams = logonService.execute(inParams);

		logger.info("forgetpasswordsubmit(HttpServletRequest request, HttpServletResponse response) method ends");
		return new ModelAndView("forgetpasswordsubmit", Constants.ERROR_MODEL, outParams);
	}
